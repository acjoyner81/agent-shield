# Tripwire Policy for AgentShield FIM
#
# Attribute flags for the severity levels below. Written out rather than using
# $(IgnoreNone)-SHa, which is how this file originally read: Tripwire 2.4 rejects
# $(...) macro references in a policy file's GLOBAL section, and IgnoreNone is
# only defined for a policy, so moving the levels to twcfg.txt failed too with
# "Configuration file uses an undefined variable: IgnoreNone".
# +p (perms), +i (inode), +n (links), +u (user), +g (group), +s (size),
# +m (mtime), +c (ctime), +C (SHA hash).
@@section GLOBAL
TWROOT=/etc/tripwire ;
TWBIN=/usr/sbin ;

SEC_CRIT   = -p+imnsugcmsC ;
SEC_CONFIG = -p+imnsugcmsC ;
SEC_APP    = -p+imnsugcmsC ;
SEC_SELF   = -p+imnsugcmsC ;

@@section FS

# Monitored AgentShield microservices and configurations
/mnt/gateway        -> $(SEC_APP) (recurse=true) ;
/mnt/java-services  -> $(SEC_APP) (recurse=true) ;
/mnt/nginx.conf     -> $(SEC_CONFIG) ;
/mnt/config         -> $(SEC_CONFIG) (recurse=true) ;

# The monitor's own policy, config, and entrypoint.
#
# Without this a tampered twpol.txt is the perfect blind spot: edit the file to
# drop a path from scope and the monitor reports clean forever while protecting
# nothing. A file integrity monitor that does not watch its own configuration is
# trivially switched off by whoever it is meant to catch.
#
# Changes here are legitimate too, so this reports them like any other object.
# The keys are deliberately not in this tree. They live in /etc/tripwire, which
# is the separate tripwire-keys volume, so that watching them would mean handing
# the container its own encryption material to read.
/mnt/tripwire       -> $(SEC_SELF) (recurse=true) ;

# Stop points. A stop point names one literal path, it does not take a pattern,
# so every generated directory has to be listed by name. A regex here compiles
# without complaint and then never matches anything, which is the quiet failure
# that left 141 build artifacts under watch.
#
# These directories are rewritten by any local test run or build while the
# source around them stays put, so monitoring them turns an ordinary rebuild
# into a mass tampering event and trains everyone to ignore this monitor.
!/mnt/gateway/__pycache__ ;
!/mnt/gateway/tests/__pycache__ ;
!/mnt/gateway/tests/e2e/__pycache__ ;
!/mnt/config/__pycache__ ;
!/mnt/config/guardrails/__pycache__ ;
!/mnt/config/guardrails/tests/__pycache__ ;
!/mnt/java-services/gateway-java/target ;
!/mnt/java-services/mcp-server/target ;
