# Tripwire Policy for AgentShield FIM
#
# Attribute flags for the severity levels below. Written out rather than using
# $(IgnoreNone)-SHa, which is how this file originally read: Tripwire 2.4 rejects
# $(...) macro references in a policy file's GLOBAL section, and IgnoreNone is
# only defined for a policy, so moving the levels to twcfg.txt failed too with
# "Configuration file uses an undefined variable: IgnoreNone".
# +p (perms), +i (inode), +n (links), +u (user), +g (group), +s (size),
# +m (mtime), +c (ctime), +C (SHA hash).
@@section GLOBAL
TWROOT=/etc/tripwire ;
TWBIN=/usr/sbin ;

SEC_CRIT   = -p+imnsugcms ;
SEC_CONFIG = -p+imnsugcms ;
SEC_APP    = -p+imnsugcms ;

@@section FS

# Monitored AgentShield microservices and configurations
/mnt/gateway        -> $(SEC_APP) (recurse=true) ;
/mnt/java-services  -> $(SEC_APP) (recurse=true) ;
/mnt/nginx.conf     -> $(SEC_CONFIG) ;
/mnt/config         -> $(SEC_CONFIG) (recurse=true) ;
